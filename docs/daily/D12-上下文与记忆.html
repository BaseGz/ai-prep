"""D12 · 上下文与记忆：短期记忆 + 裁剪策略"""
import sys
from pathlib import Path

DAY09 = Path(__file__).resolve().parents[1] / "day09-handwritten-agent"
sys.path.insert(0, str(DAY09))

from langchain.agents import AgentState, create_agent
from langchain.agents.middleware import before_model
from langchain_core.messages import RemoveMessage
from langgraph.checkpoint.memory import InMemorySaver
from langgraph.graph.message import REMOVE_ALL_MESSAGES

from agent_loop import model

QUESTIONS = ["我叫 Base", "我在学 LangChain", "我在准备实习", "我叫什么？"]
SYSTEM = "你是一个简洁的助手，回答不超过两句话。"


def chat(agent, thread_id, questions):
    """在同一个 thread 里连问几句，打印每句的答复和累积的消息条数"""
    cfg = {"configurable": {"thread_id": thread_id}}
    for q in questions:
        r = agent.invoke({"messages": [{"role": "user", "content": q}]}, cfg)
        print(f"  问「{q}」")
        print(f"  答：{r['messages'][-1].content[:60]}")
        print(f"  messages 条数 = {len(r['messages'])}")
    print()


def keep_last_n(n):
    """只保留最近 n 条消息的中间件"""

    @before_model
    def trim(state: AgentState, runtime) -> dict | None:
        msgs = state["messages"]
        if len(msgs) <= n:
            return None
        return {"messages": [RemoveMessage(id=REMOVE_ALL_MESSAGES), *msgs[-n:]]}

    return trim


if __name__ == "__main__":
    print("=== 实验 1：没有 checkpointer ===")
    a1 = create_agent(model, [], system_prompt=SYSTEM)
    for q in ["我叫 Base", "我叫什么？"]:
        r = a1.invoke({"messages": [{"role": "user", "content": q}]})
        print(f"  问「{q}」-> {r['messages'][-1].content[:50]} | messages={len(r['messages'])}")

    print()
    print("=== 实验 2：checkpointer + 同一个 thread_id ===")
    a2 = create_agent(model, [], system_prompt=SYSTEM, checkpointer=InMemorySaver())
    chat(a2, "u1", QUESTIONS)

    print("=== 实验 3：换一个 thread_id 问同样的话 ===")
    chat(a2, "u2", ["我叫什么？"])

    print("=== 实验 4：裁剪到最近 4 条 ===")
    a3 = create_agent(
        model, [], system_prompt=SYSTEM,
        middleware=[keep_last_n(4)], checkpointer=InMemorySaver(),
    )
    chat(a3, "t1", QUESTIONS)

    print("=== 实验 5：裁剪到最近 6 条 ===")
    a4 = create_agent(
        model, [], system_prompt=SYSTEM,
        middleware=[keep_last_n(6)], checkpointer=InMemorySaver(),
    )
    chat(a4, "t2", QUESTIONS)

4.2 五个实验，我实跑到的结果

#条件实测说明什么
1 没有 checkpointer messages=2（两次都是）； 问「我叫什么」答「我不知道你叫什么，因为你还没有告诉我」 每次 invoke 都是全新的 state。D9 那套「自己 append 到 messages」的思路， 在框架版里没有 checkpointer 就等于没接上
2 checkpointer + 同一个 thread_id 条数 2 → 4 → 6 → 8，每轮 +2； 第 4 轮答「你叫 Base」 这就是 README 要的「记住 3 轮前」。记忆 = 状态累积，不是模型记住了
3 换 thread_id messages=2，答「抱歉，我不知道你的名字」 thread 是隔离边界。同一个 agent 对象、同一个 checkpointer，换个 id 就是另一个人的对话
4 裁到最近 4 条 条数 2 → 4 → 5 → 5；第 4 轮答「抱歉，我不知道你的名字 —— 你还没有告诉过我」 裁太狠：第 1 轮那句「我叫 Base」被裁掉了。注意第 3 轮开始条数就不再往上涨（5 是上限 +1）
5 裁到最近 6 条 条数 2 → 4 → 6 → 7；第 4 轮答「你叫 Base」 裁到 6 就够：第 4 轮时第 1 轮还在窗口里。这就是「上下文工程」要的那个界

「你的活」三处（必须自己改，改完重跑）

  1. 找出临界值。实验 4 用 4 记不住、实验 5 用 6 记得住 —— 那就把 keep_last_n(5) 单独跑一次，看它落在哪一边。 判定标准：能说出「记住第 1 轮」最少要保留几条，并解释为什么是偶数（一问一答算两条）。
  2. 改成「永远保留第一条 + 最近 n 条」。官方文档的 trim 例子就是 [first_msg] + recent_messages —— 它没有把第一条裁掉。照这个改 keep_last_n， 然后用 n=4 再跑一次。 判定标准：改完之后 n=4 也能答出名字了 —— 因为「我叫 Base」正好是第一条。 但这是取巧：如果关键信息出现在第 2 轮而不是第 1 轮，这个策略照样会丢。想一想为什么。
  3. （选做）用 SummarizationMiddleware 替代手写裁剪。 把 middleware=[keep_last_n(4)] 换成： middleware=[SummarizationMiddleware(model=model, trigger=("tokens", 500), keep=("messages", 2))] （记得从 langchain.agents.middleware 导入）。 判定标准：观察两件事 —— ① 摘要之后消息条数变成几条；② 第 4 轮还记不记得名字。 摘要和裁剪的本质差别：裁剪是「丢」，摘要是「花一次 LLM 调用换一段压缩后的记忆」。

第 1 次：代码

勾选：week2-tools-agent/day12-memory/memory.py

feat(day12): 短期记忆，能记住 3 轮前提到的信息

现象：
同一句「我叫什么」，不加 checkpointer 时回答「你没有告诉过我」，
加了 checkpointer 并固定 thread_id 之后能答出「Base」，messages 条数从 2 一路涨到 8。

我以为：
记忆是模型的能力 —— 把历史塞进 messages 它就能一直记得。

实际：
每次 invoke 都是一个全新的 state，模型看不到上一次的消息；是 checkpointer 把 state 存下来、
下次按 thread_id 取回来再拼进 messages。记忆是图的属性，不是模型的属性。
而裁剪到 4 条时立刻就忘了（第 1 轮那句被裁掉），裁到 6 条才刚好记住。

做法：
create_agent(checkpointer=InMemorySaver())，每次 invoke 传
{"configurable": {"thread_id": "u1"}}；
用 @before_model + RemoveMessage(REMOVE_ALL_MESSAGES) 手写裁剪，分别试 4 / 6 条；
换 thread_id 验证隔离。

结论：
上下文工程没有标准答案，只有实测出来的界 —— 记住 3 轮前至少得留 6 条，
少一条就失忆。而这个界随对话长度变化，所以它必须是个运行时策略，不能写死。

第 2 次：文档

勾选：docs/journal/2026-10-02.md + docs/decisions.md + docs/progress.md + README.md + docs/daily/D12-上下文与记忆.html

docs(day12): 短期记忆的实测边界 —— 留 6 条记得住，留 4 条就忘

现象：
裁剪中间件保留最近 4 条时，第 3 轮开始 messages 稳定在 5 条不再增长，
第 4 轮问名字答「你还没有告诉过我」；改成保留 6 条后条数是 7，答出了名字。

我以为：
留得越多越稳，成本和效果是这条连续曲线上的取舍。

实际：
不是连续曲线，是台阶 —— 关键是「第一轮那句还在不在窗口里」。
一问一答算两条，记住第 1 轮至少留 6 条，5 条就已经丢了。

做法：
把 4 / 5 / 6 三个值各跑一遍定出临界点；
再按官方文档的写法改成「永远保留第一条 + 最近 n 条」，验证 n=4 也能记住。

结论：
裁剪策略的价值不在「留几条」，在「留哪些」—— 第一条值得特殊对待，
但那只在关键信息出现在开头时成立。换了场景就得重新定策略，没有通用参数。

6坑预警

你会看到真实原因
ValueError: Checkpointer requires one or more of the following 'configurable' keys: thread_id, ... 建 agent 时传了 checkpointer，但 invoke 时没传第二个参数。这个报错很友好， 有些版本不报错，直接在每次 invoke 里新建一个 thread，表现为「记忆完全没生效」
明明加了 checkpointer，还是记不住 三种可能：① 每次 thread_id 都不一样（比如用了 uuid4()）； ② 每次都重新 create_agent 且每次都 new 一个 InMemorySaver()； ③ 只把新消息传进去、期望它自己合并 —— checkpointer 保证的是「读到旧 state」， 而 messages 是靠 add_messages reducer 做追加
裁到 6 条，messages 条数却是 7 不是 6 裁剪发生在本轮 AI 回复之前（@before_model）。裁完之后这一轮又追加 一问一答，所以条数 = 上限 + 1 或 + 2。实验 4 里稳定在 5 就是这个原因
裁剪之后 API 报 400，提示 tool 消息不匹配 把带 tool_calls 的 AIMessage 裁掉了，但它的 ToolMessage 还在。 官方文档 Delete messages 末尾专门警告了这条 —— 消息历史必须合法， 而「合法」是按供应商的规矩算的
跑完脚本，重开一次终端，记忆全没了 InMemorySaver 存在进程内存里。这不是 bug，是它的定义。 要跨进程保留，得换 SqliteSaver / PostgresSaver —— 那是 D13 的事
实验 2 的答复措辞每次都不一样 正常。看条数和答没答出名字，别看它怎么措辞
