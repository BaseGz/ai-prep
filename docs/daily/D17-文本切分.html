"""D17 · 文本切分：同一份文档，四种粒度各切一遍"""
from pathlib import Path

from langchain_community.document_loaders import BSHTMLLoader
from langchain_core.documents import Document
from langchain_text_splitters import RecursiveCharacterTextSplitter

ROOT = Path(__file__).resolve().parents[2]
HTML_DIR = ROOT / "datasets/raw/x4-manual"
MD = ROOT / "docs/journal/2026-10-05.md"


def load_all_html():
    """加载 114 个 HTML，并补上 section（昨天欠的那一步，这次要返回 docs）"""
    docs = []
    for f in sorted(HTML_DIR.rglob("*.html")):
        section = f.relative_to(HTML_DIR).parts[0]
        for d in BSHTMLLoader(str(f), open_encoding="utf-8").load():
            d.metadata["section"] = section
            docs.append(d)
    return docs


def report(chunks, label):
    """一组 chunk 的统计：条数 / 平均 / 最短 / 最长"""
    lens = [len(c.page_content) for c in chunks]
    avg = sum(lens) // len(lens)
    print(f"{label:28} {len(chunks):>4} 块 | 平均 {avg:>4} | 最短 {min(lens):>4} | 最长 {max(lens):>4}")


def main():
    md_text = MD.read_text(encoding="utf-8")
    doc = Document(page_content=md_text, metadata={"source": MD.name, "section": "journal"})
    print(f"源文档：{MD.name}，{len(md_text)} 字\n")

    for size in (300, 500, 1000, 2000):
        sp = RecursiveCharacterTextSplitter(chunk_size=size, chunk_overlap=0)
        report(sp.split_documents([doc]), f"size={size} overlap=0")

    print()
    for size, pct in ((500, 0.15), (1000, 0.15), (1000, 0.3)):
        ov = int(size * pct)
        sp = RecursiveCharacterTextSplitter(chunk_size=size, chunk_overlap=ov)
        report(sp.split_documents([doc]), f"size={size} overlap={ov}")


if __name__ == "__main__":
    main()

4.2 三处「你的活」

① cut_ratio(chunks) —— 写「有没有把一句话切断」的判据

执行单要求你记录「是否把一句话切断」，但「切断」得先被定义成一件事，代码才能数。 一个够用的判据：块的结尾如果不是句末标点，就算切断了。填空：

ENDS = ("。", "！", "？", "；", "\n", "”", "」")

def cut_ratio(chunks):
    """返回 (被切断的块数, 总块数_不含最后一块)"""
    cut = 0
    for c in chunks[____]:          # ① 最后一块没有「后面」可以接，排除掉
        text = c.page_content.____()   # ② 去掉尾部空白，否则 '\n' 会被当成正常结尾
        if not text.endswith(____):    # ③ 填判据
            cut += 1
    return cut, len(chunks) - 1

③ try_token_split() —— 字符 vs token，差多少

def try_token_split():
    """同样「1000 的大小」，按 token 量会得到不同的块"""
    from langchain_text_splitters import RecursiveCharacterTextSplitter
    import tiktoken

    text = MD.read_text(encoding="utf-8")
    enc = tiktoken.get_encoding("cl100k_base")
    n_tok = len(enc.encode(text))
    print(f"{len(text)} 字 = {n_tok} token，1 token 约 {len(text)/n_tok:.2f} 字")

    sp = RecursiveCharacterTextSplitter.from_tiktoken_encoder(
        encoding_name="cl100k_base", chunk_size=256, chunk_overlap=32
    )
    chunks = sp.split_text(text)
    print(f"token 切分 256/32 -> {len(chunks)} 块，平均 {sum(len(c) for c in chunks)//len(chunks)} 字符")

预期：2725 字 = 2099 token，1 token ≈ 1.3 字；切出 11 块，平均 251 字符。 中文比英文「字少 token 多」，所以按字符切的 1000，实际可能是 700–800 个 token —— 这就是官方说 「字符数只是代理指标」的意思。

6坑预警

症状真实原因
AttributeError: 'str' object has no attribute 'page_content' split_text() 返回 list[str]，split_documents() 返回 list[Document]。 统计函数混用两种返回值就会炸（我写骨架时就撞了一次）
切完 metadata 里的 section 没了 用了 split_text()。它只认字符串，不认 Document，metadata 全丢。 要保留就 split_documents()
中文被从句子中间硬切开 默认 separators 是 ["\n\n","\n"," ",""]，中文没空格， 段落之外会一路退到 ""（字符级）硬切。实测加中文标点作为分隔符效果有限， 真要优化得配 is_separator_regex
块数暴涨、内容大量重复 chunk_overlap 接近 chunk_size 了。实测 size=100/overlap=100 会切出 49 块。 重叠保持在 10%–20%
UnicodeDecodeError: 'gbk' codec ... 昨天那条又来了。HTML 一律 BSHTMLLoader(path, open_encoding="utf-8")
向量库报 Expected metadata value to be a str, int, float or bool D18 才会撞。metadata 里塞了 list（比如 JSON 的 tags）。 入向量库前统一转成 ",".join(...)
