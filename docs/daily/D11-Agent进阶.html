"""D11 · Agent 进阶用法：4 个工具 + 结构化输出 + 错误处理"""
import sys
from pathlib import Path

DAY09 = Path(__file__).resolve().parents[1] / "day09-handwritten-agent"
sys.path.insert(0, str(DAY09))

from langchain.agents import create_agent
from langchain.agents.middleware import ToolRetryMiddleware
from langchain_core.tools import tool
from pydantic import BaseModel, Field

from agent_loop import find_order, track_package, model

MOCK_COUPONS = {
    "8899": ["满 100 减 15（有效至 2026-11-30）", "免运费券 ×1"],
    "6666": [],
}


@tool
def query_coupon(phone_tail: str) -> str:
    """查询某个手机号尾号名下可用的优惠券。

    Args:
        phone_tail: 手机号后四位，例如 8899
    """
    coupons = MOCK_COUPONS.get(phone_tail)
    if coupons is None:
        return f"没有找到尾号 {phone_tail} 的账户"
    if not coupons:
        return f"尾号 {phone_tail} 名下暂无可用优惠券"
    return "；".join(coupons)


@tool
def query_refund(order_id: str) -> str:
    """查询某个订单的退款进度（本工具故意抛异常，用来演示错误处理）。

    Args:
        order_id: 订单号，例如 ORD-20261015-001
    """
    raise RuntimeError("退款服务暂时不可用（下游 503）")


class OrderAnswer(BaseModel):
    """订单状态的结构化结果"""

    order_id: str = Field(description="订单号")
    status: str = Field(description="订单当前状态")
    summary: str = Field(description="一句话总结，不超过 30 字")


def build_agent(with_retry: bool = False):
    middleware = [ToolRetryMiddleware(max_retries=1, initial_delay=0.1)] if with_retry else []
    return create_agent(
        model,
        [find_order, track_package, query_coupon, query_refund],
        system_prompt="你是一个物流智能查询助手",
        response_format=OrderAnswer,
        middleware=middleware,
    )


def ask(agent, question):
    result = agent.invoke({"messages": [{"role": "user", "content": question}]})
    print("消息条数：", len(result["messages"]))
    for i, m in enumerate(result["messages"], 1):
        calls = [c["name"] for c in getattr(m, "tool_calls", []) or []]
        print(f"{i}. {type(m).__name__} 调用={calls or '-'} | {str(m.content)[:58]}")
    print("结构化结果：", result.get("structured_response"))
    return result


if __name__ == "__main__":
    print("=== 实验 1：4 个工具 + 结构化输出 ===")
    ask(build_agent(), "尾号 8899 的订单现在什么状态？挑最新那单说")
    print()
    print("=== 实验 2：问到会抛异常的工具（没有 middleware）===")
    try:
        ask(build_agent(), "订单 ORD-20261015-001 的退款进度怎么样了？")
    except Exception as e:
        print(">>> 中断：", type(e).__name__, "|", str(e)[:120])
    print()
    print("=== 实验 3：加上 ToolRetryMiddleware 再问一次 ===")
    ask(build_agent(with_retry=True), "订单 ORD-20261015-001 的退款进度怎么样了？")

#看点我实跑到的结果（你跑出来不必逐字一样）
1 结构化输出长什么样 7 条消息，最后两条是 AIMessage 调用=['OrderAnswer'] + ToolMessage | Returning structured response: ...； 结果在 structured_response，不在 messages 里
2 裸工具抛异常会怎样 >>> 中断： RuntimeError | 退款服务暂时不可用（下游 503） —— 一条消息都没拿到，整轮全丢
3 加上 middleware 之后 条数不固定（我两次分别跑到 7 条和 9 条）。稳定出现的是第 3 条： Tool 'query_refund' failed after 2 attempts with RuntimeError ... Please try again. —— 之后模型要么直接转去查物流（7 条），要么顺着那句 Please try again 再调一次 query_refund、再失败一次才转物流（9 条）。最终 status 都会写成「退款查询失败」

「你的活」三处（改完重跑）

  1. query_coupon 的三种情况：骨架里三个分支已经分好了，这条不用改代码，改成验收 —— 在 __main__ 末尾临时加两行 print(query_coupon.invoke({"phone_tail": ...}))， 分别问 6666（有账户没券）和 1234（没这个账户）。 判定标准：两句话不能是同一句。如果都答「暂无可用优惠券」，就是 D9「查无此单 / 有单未发货」那个坑在这个工具上重演。
  2. ToolRetryMiddleware 的 max_retries：改成 0 跑一次、改成 3 跑一次（跑完改回 1）， 抄下第 3 条 ToolMessage 里的 after N attempt(s)。 实测：0 → after 1 attempt（单数，没有 s），1 → after 2 attempts， 3 → after 4 attempts。说明 max_retries 是重试次数，总次数 = 1 + max_retries。 ⚠️ 但 max_retries=0 那次会出现两次 query_refund 调用：中间件只试了 1 次， 模型读完文案结尾的 Please try again. 自己又调了一次。数带 query_refund 的 AIMessage 有几个就能看出来。
  3. （接上 D10 那条坑）在 system_prompt 末尾加一句： 「如果某个工具失败，先尝试用其它工具回答，最后说明哪部分没查到。」再跑实验 3， 对比 structured_response 的 status 和 summary。 实测：status 从「退款进度查询失败」变成「退款进度未知（物流已签收）」。 这一条是 D8「工具说得含糊模型就替你补」的进阶版：现在是「你告诉它怎么补，它就怎么补」 —— 代价是物流事实被塞进了退款字段，字段语义被污染。

feat(day11): Agent 接入结构化输出，结果落在 structured_response

现象：
加了 response_format 之后，messages 最后一条不再是 AIMessage，
而是一条内容为 "Returning structured response: ..." 的 ToolMessage。

我以为：
结构化输出是让模型直接吐一段 JSON 字符串，我从 content 里 json.loads 解析。

实际：
框架把它做成了一次「假工具调用」—— 最后那条 AIMessage 的 tool_calls 里，
工具名就叫 OrderAnswer（我的 Pydantic 类名）。schema 就是这个工具的参数表。

做法：
response_format=OrderAnswer（Pydantic）；先试过 ProviderStrategy，
DeepSeek 直接 400 This response_format type is unavailable now，只能走 ToolStrategy。

结论：
结构化输出 = 给模型一个只有 schema、没有实现的工具，逼它按这个形状填参数；
结果在 result["structured_response"]，不在 messages 里。

第 2 条：错误处理

勾选：agent_advanced.py + docs/decisions.md + docs/journal/2026-10-01.md + docs/progress.md + README.md

fix(day11): 工具抛异常会中断整轮，改用 ToolRetryMiddleware 兜住

现象：
裸 @tool raise RuntimeError 之后，agent.invoke 直接把异常抛到顶层，
前面已经跑过的几轮消息一条都拿不到，structured_response 也没有。

我以为：
框架会像 D9 我手写的那样，把工具异常包成 ToolMessage 回传，循环继续。
README 里也写着「执行工具那一步没有 try/except」—— 我原以为补上就行。

实际：
create_agent 默认不兜工具异常；兜底是 ToolRetryMiddleware，
它默认 on_failure='continue'，重试完把错误包成一条 ToolMessage 交给模型。

做法：
middleware=[ToolRetryMiddleware(max_retries=1)]；max_retries=1 的文案是
failed after 2 attempts（1 次原始 + 1 次重试）。

结论：
工具失败不是异常，是一条「坏消息」—— 把错误当结果回传，模型自己会找替代路径
（实测它失败后转去查了物流，最后在 status 里写「退款查询失败」）。

6坑预警

你会看到真实原因
RuntimeError: 退款服务暂时不可用（下游 503） 直接冒到顶层 这不是你的 bug，是实验 2 的预期结果。create_agent 默认不捕获工具异常。 兜底要显式加 ToolRetryMiddleware（或在工具内部 try/except）
400 This response_format type is unavailable now 用了 ProviderStrategy。DeepSeek 不支持模型原生结构化输出， 只能用 ToolStrategy（或直接传 Pydantic 类，默认就走 ToolStrategy）
400 Thinking mode does not support this tool_choice 模型名用了 deepseek-flash。这条从 D6 踩到今天，工具调用/结构化输出一律 deepseek-chat
max_retries=1，但文案写 after 2 attempts 不是 off-by-one：max_retries 是「重试几次」，总调用次数 = 1 + max_retries
想从 result["messages"][-1].content 拿结构化结果，拿到一串 Returning structured response: ... 结果在 result["structured_response"]，是 OrderAnswer 实例，可以直接 .order_id； message 里那条只是给模型看的收尾文本
加了 response_format 后，模型先调工具、最后才填 schema 正常。结构化输出发生在最后一轮，前面该调几个工具照调。 所以「4 个工具 + 结构化」不冲突，是叠加关系
去官方 Agents 页找「多工具」小节，找不到 确实没有。tools= 传列表就是多工具，官方没有单独展开这一节
实验 3 跑两次，消息条数不一样（一次 7 条、一次 9 条） 不是随机 bug。middleware 生成的错误文案结尾是 Please try again.， 模型照着这句话又调了一次同一个工具。想让它别重试：改 system_prompt， 或者干脆在工具内部自己 try/except、返回你能控制的错误串
