3.1 建文件，整段粘贴

"""D14 · 多功能智能助手 v0.2：工具 + 记忆 + 流式"""
import ast
import operator
import sqlite3
import sys
from pathlib import Path

DAY09 = Path(__file__).resolve().parents[1] / "day09-handwritten-agent"
sys.path.insert(0, str(DAY09))

from langchain.agents import create_agent
from langchain_core.messages import AIMessageChunk
from langchain_core.tools import tool
from langgraph.checkpoint.sqlite import SqliteSaver

from agent_loop import find_order, model, track_package

DB = Path(__file__).with_name("assistant.db")
THREAD = "cli"

SYSTEM = """你是影石的内部助手，能查订单、查物流、算数、记笔记。
回答简洁；需要数据时必须调工具，不要凭记忆编造。"""


@tool(parse_docstring=True)
def calc(expression: str) -> str:
    """计算一个四则运算表达式

    Args:
        expression: 只含数字和 + - * / ( ) 的表达式，例如 (12+8)*3
    """
    ops = {
        ast.Add: operator.add,
        ast.Sub: operator.sub,
        ast.Mult: operator.mul,
        ast.Div: operator.truediv,
        ast.USub: operator.neg,
    }

    def ev(node):
        if isinstance(node, ast.Constant) and isinstance(node.value, (int, float)):
            return node.value
        if isinstance(node, ast.BinOp) and type(node.op) in ops:
            return ops[type(node.op)](ev(node.left), ev(node.right))
        if isinstance(node, ast.UnaryOp) and type(node.op) in ops:
            return ops[type(node.op)](ev(node.operand))
        raise ValueError("只支持数字与 + - * / ( )")

    try:
        return str(ev(ast.parse(expression, mode="eval").body))
    except Exception as e:
        return f"算不了：{e}"


@tool(parse_docstring=True)
def save_note(title: str, content: str) -> str:
    """把一段内容记成笔记文件

    Args:
        title: 笔记标题，会用作文件名
        content: 笔记正文
    """
    safe = "".join(c for c in title if c.isalnum() or c in "-_ ").strip() or "note"
    path = Path(__file__).with_name("notes") / f"{safe}.md"
    path.parent.mkdir(exist_ok=True)
    path.write_text(content, encoding="utf-8")
    return f"已写入 {path.name}（{len(content)} 字）"


TOOLS = [find_order, track_package, calc, save_note]


def build():
    conn = sqlite3.connect(DB, check_same_thread=False)
    agent = create_agent(
        model, TOOLS, system_prompt=SYSTEM, checkpointer=SqliteSaver(conn)
    )
    return agent, conn


def main():
    agent, conn = build()
    cfg = {"configurable": {"thread_id": THREAD}}
    print("助手已启动（/exit 退出）")
    while True:
        q = input("\n你> ").strip()
        if q in {"/exit", "/quit"}:
            break
        if not q:
            continue
        print("助手> ", end="", flush=True)
        for chunk, _meta in agent.stream(
            {"messages": [{"role": "user", "content": q}]},
            cfg,
            stream_mode="messages",
        ):
            if isinstance(chunk, AIMessageChunk) and chunk.content:
                print(chunk.content, end="", flush=True)
        print()
    conn.close()


if __name__ == "__main__":
    main()
为什么 calc 要用 ast 而不是 eval()： eval 会把字符串当代码执行，模型被诱导一下就能跑 __import__('os').system(...)。 ast.parse 先把表达式拆成语法树，只放行数字和四则运算符，遇到别的节点直接抛错 —— 工具的参数是模型写的，等于外部输入，不能直接喂给 eval。

6坑预警

症状真实原因
回答里混进了 ToolMessage(...) 或工具原始返回值 stream(mode="messages") 会把工具返回的消息也当 chunk 吐出来， 必须用 isinstance(chunk, AIMessageChunk) 过滤，只打模型说的话
重开之后完全不记得 三种可能：thread_id 变了、checkpointer 没传、或者你在用 InMemorySaver。先打印启动时的 checkpoint 数，一眼分清
模型死活不调工具，自己瞎答 docstring 只写了「做什么」没写「什么时候用」。把 Args: 里的参数说明写具体， SYSTEM 里加一句「需要数据必须调工具」
calc 报「算不了：invalid syntax」 模型把「128 加 72 乘以 3」原样传进来了，没转成算式。参数描述里要写死例子，例如 只含数字和 + - * / ( )，例如 (12+8)*3
save_note 报路径错误 / 文件写到奇怪的地方 标题里带了 / 或 :。代码里已经用白名单过滤，如果你改动了这段，注意别去掉
输入中文显示成乱码 PyCharm 运行窗口编码。在运行配置的 环境变量 里加 PYTHONIOENCODING=utf-8
ModuleNotFoundError: agent_loop 文件没建在 day14-assistant/ 下，或者 PyCharm 解释器不是项目 venv
