发现一句话
模型会编造像真的值输入里没提年龄，它给 age=30；输入「请介绍一下这家公司」连人都没有，它照样返回一个填满的 Person。比填 0 危险 —— 0 一眼可疑，30 不像

继承 BaseModel 不是「通过审查」是「拿到造表的工具」。不继承不报错，只是静默给你一份空 Schema → invoke 回来一个 {}

结论行建议（或你自己的话）：「跑通了」≠「跑对了」，「没报错」更不等于「对」—— 一天之内撞到 5 个坑，其中 3 个是不报错的。

"""D7 · 流式对话 CLI v0.1

把前 6 天串成一个能用的命令行工具：
- D1  模型创建      init_chat_model
- D3  流式输出      model.stream()
- D4  多轮对话      自己维护 messages 数组（模型没有记忆，历史每轮重发）
- D5  LCEL 管道     prompt | model | parser
- D6  结构化抽取    with_structured_output(Lead) → 直接驱动代码分支

命令：
  普通文字        → 流式对话（带多轮上下文）
  /extract 文字   → 结构化抽取客户线索 + 分支判断
  /history        → 看历史有几条
  /clear          → 清空历史
  /exit           → 退出
"""
from __future__ import annotations

import os
from pathlib import Path

from dotenv import load_dotenv
from langchain.chat_models import init_chat_model
from langchain_core.messages import AIMessage, HumanMessage, SystemMessage
from pydantic import BaseModel, Field
from rich.console import Console
from rich.panel import Panel
from rich.table import Table

REPO_ROOT = Path(__file__).resolve().parents[2]
load_dotenv(REPO_ROOT / ".env")

# ★ 固定用 deepseek-chat。deepseek-flash 是 thinking 模式，
#   结构化输出底层是 tool_choice，会直接 400（D6 实测）
MODEL_NAME = "deepseek:deepseek-chat"
BASE_URL = os.getenv("DEEPSEEK_BASE_URL")
API_KEY = os.getenv("DEEPSEEK_API_KEY")

SYSTEM_PROMPT = "你是一个简洁的助手，回答尽量短，不要啰嗦。"

console = Console()


class Lead(BaseModel):
    """从一句话里抽出来的客户线索"""

    name: str = Field(description="姓名")
    company: str = Field(description="公司")
    need: str = Field(description="需求")
    urgency: str = Field(description="紧急度，只能是 高 / 中 / 低 之一")


def build_model():
    """D1 的活：把模型建出来。"""
    return init_chat_model(
        MODEL_NAME,
        base_url=BASE_URL,
        api_key=API_KEY,
        temperature=0,
    )


def stream_reply(model, messages: list) -> None:
    """D3 的活：流式打印，并把完整回复追加进历史。

    注意这里是 print 而不是 console.print —— 流式要精确控制每个 chunk 的尾巴，
    原生 print(end="", flush=True) 最省心。
    """
    print("助手 > ", end="", flush=True)
    pieces: list[str] = []
    for chunk in model.stream(messages):
        piece = chunk.content if isinstance(chunk.content, str) else ""
        pieces.append(piece)
        print(piece, end="", flush=True)
    print()
    # ★ D4 的关键：把这一轮回复补进历史，下一轮才「记得」
    messages.append(AIMessage("".join(pieces)))


def show_lead(lead: Lead) -> None:
    """D6 的活：拿到的是对象，不是一段文字 —— 所以能直接 if 分支。"""
    table = Table(title="抽取结果", show_header=False, border_style="cyan")
    table.add_row("姓名", lead.name)
    table.add_row("公司", lead.company)
    table.add_row("需求", lead.need)
    table.add_row("紧急度", lead.urgency)
    console.print(table)

    if lead.urgency == "高":
        console.print(f"[bold red]【立即跟进】{lead.name}（{lead.company}）→ 转销售主管[/bold red]")
    elif lead.urgency == "中":
        console.print(f"[bold yellow]【今日回复】{lead.name}（{lead.company}）→ 普通工单[/bold yellow]")
    else:
        console.print(f"[green]【入池培育】{lead.name}（{lead.company}）→ 定期触达[/green]")


def main() -> None:
    model = build_model()
    extractor = model.with_structured_output(Lead)  # 建一次，反复用
    messages: list = [SystemMessage(SYSTEM_PROMPT)]

    console.print(
        Panel.fit(
            "流式对话 CLI v0.1\n"
            "普通文字直接聊 ／ [cyan]/extract 一句话[/cyan] 抽线索\n"
            "[cyan]/history[/cyan] 看历史 ／ [cyan]/clear[/cyan] 清空 ／ [cyan]/exit[/cyan] 退出",
            border_style="cyan",
        )
    )

    while True:
        try:
            line = input("你 > ").strip()
        except (EOFError, KeyboardInterrupt):
            console.print()
            break

        if not line:
            continue
        if line in ("/exit", "/quit"):
            break
        if line == "/clear":
            messages = [SystemMessage(SYSTEM_PROMPT)]
            console.print("[yellow]历史已清空[/yellow]")
            continue
        if line == "/history":
            console.print(f"[yellow]历史共 {len(messages)} 条（含 1 条 system）[/yellow]")
            continue
        if line.startswith("/extract"):
            text = line[len("/extract") :].strip()
            if not text:
                console.print("[red]用法：/extract 一句话[/red]")
                continue
            show_lead(extractor.invoke(text))
            continue

        # 普通对话：先补进用户这句，再流式要回复
        messages.append(HumanMessage(line))
        stream_reply(model, messages)

    console.print("[cyan]再见[/cyan]")


if __name__ == "__main__":
    main()
